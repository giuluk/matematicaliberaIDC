\newcommand{\iperboleato}[4]{%Iperbole con asse trasverso orizzontale
  % Sintassi: \iperbole{a}{b}{domain}{colore}
  % Sintassi: \iperbole{3}{4}{Maroon}
  \draw[thick,color=#4] plot[domain=-#3:+#3] 
       ({#1*cosh(\x)},{#2*sinh(\x)});
  \draw[thick,color=#4] plot[domain=-#3:+#3] 
       ({-#1*cosh(\x)},{#2*sinh(\x)});
}

\newcommand{\iperboleatv}[4]{%Iperbole con asse trasverso verticale
  % Sintassi: \iperbole{a}{b}{domain}{colore}
  % Sintassi: \iperbole{3}{4}{Maroon}
  \draw[thick,color=#4] plot[domain=-#3:+#3] 
       ({#1*sinh(\x)}, {#2*cosh(\x)});
  \draw[thick,color=#4] plot[domain=-#3:+#3] 
       ({#1*sinh(\x)}, {-#2*cosh(\x)});
}
